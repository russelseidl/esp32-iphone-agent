# Bench Harness Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Choose mirror and pointer settings from measurements: a `bench` tool runs trials in a scratch Freeform board, measures tap error and step latency per setting, logs every trial to telemetry and writes a ranked summary.

**Architecture:** Spec section 4 of `docs/superpowers/specs/2026-10-19-measured-navigation-design.md`. A pure module `mac/bench.py` holds what can be tested without a phone: dot detection on a white canvas (dark blobs, matched to the planned grid by nearest neighbour), trial records, per-setting statistics (mean error, p50/p95 latency, drops), winner selection with margins, and the Markdown summary. The server's `bench` tool orchestrates: apply a setting (`res`, `fps`, `coex`, `pace`, aiming mode, calibration model), recalibrate when the setting changes the pointer map, tap N planned points, read the dots back, record `trial` events, restore the baseline, and write `~/.esp32-agent/bench/<date>.md`.

**Tech Stack:** Python 3.13, Pillow, pytest via `uv run --with`. No firmware change (`pace`, `res`, `fps`, `coex` exist).

```python
"""Measurement side of the A/B bench: dots on a white canvas, per-trial records, statistics and the summary.

Score: a setting wins its axis on tap error when errors differ by more than ERROR_MARGIN units; otherwise on
median latency when that differs by more than LATENCY_MARGIN; otherwise the baseline (first listed) stays.
"""
from dataclasses import dataclass, asdict
from statistics import mean, median

from pointer import Frame, _blobs

CANVAS_WHITE = 0.80     # share of pixels above WHITE_LEVEL for a frame to count as a blank canvas
WHITE_LEVEL = 225
DOT_LEVEL = 120
DOT_MIN, DOT_MAX = 6, 900   # blob size bounds in pixels at the frame's scale
ERROR_MARGIN = 3.0
LATENCY_MARGIN = 150


def find_dots(frame: Frame) -> list[tuple[int, int]] | None:
    """Centres (0-1000) of dark dots on a mostly white frame; None if the frame is not a canvas."""
    g = frame.gray
    white = sum(1 for v in g if v > WHITE_LEVEL) / len(g)
    if white < CANVAS_WHITE:
        return None
    pts = [(i % frame.w, i // frame.w) for i, v in enumerate(g) if v < DOT_LEVEL]
    out = []
    for bl in _blobs(pts):
        if DOT_MIN <= len(bl) <= DOT_MAX:
            out.append((round((sum(p[0] for p in bl) / len(bl) + 0.5) * 1000 / frame.w),
                        round((sum(p[1] for p in bl) / len(bl) + 0.5) * 1000 / frame.h)))
    return out


def match_dots(planned: list[tuple[int, int]], found: list[tuple[int, int]], max_dist: float = 120) -> list[float | None]:
    """Error per planned tap (distance to its nearest unused dot) or None if no dot landed within max_dist."""
    free = list(found)
    errors: list[float | None] = []
    for px, py in planned:
        best = None
        for d in free:
            dist = ((d[0] - px) ** 2 + (d[1] - py) ** 2) ** 0.5
            if dist <= max_dist and (best is None or dist < best[0]):
                best = (dist, d)
        if best is None:
            errors.append(None)
        else:
            free.remove(best[1])
            errors.append(round(best[0], 1))
    return errors


@dataclass
class Trial:
    axis: str
    setting: str
    error: float | None       # units; None = the tap left no dot where expected
    latency_ms: float         # command round trip plus wait for a fresh frame
    drops: int = 0            # BLE video drops during the trial

    def row(self) -> dict:
        return {"kind": "trial", **asdict(self)}


def summarize(trials: list[Trial]) -> dict:
    out: dict = {}
    for t in trials:
        s = out.setdefault(t.axis, {}).setdefault(t.setting, {"errors": [], "latencies": [], "drops": 0, "misses": 0})
        if t.error is None:
            s["misses"] += 1
        else:
            s["errors"].append(t.error)
        s["latencies"].append(t.latency_ms)
        s["drops"] += t.drops
    for axis in out.values():
        for s in axis.values():
            s["n"] = len(s["latencies"])
            s["error_mean"] = round(mean(s["errors"]), 1) if s["errors"] else None
            lat = sorted(s["latencies"])
            s["latency_p50"] = median(lat)
            s["latency_p95"] = lat[min(len(lat) - 1, int(0.95 * len(lat)))]
            del s["errors"], s["latencies"]
    return out


def pick_winners(summary: dict) -> dict:
    winners = {}
    for axis, settings in summary.items():
        names = list(settings)
        base = names[0]
        best, by = base, "baseline"
        for name in names[1:]:
            a, b = settings[best], settings[name]
            ea, eb = a["error_mean"], b["error_mean"]
            if eb is None:
                continue
            if ea is None or ea - eb > ERROR_MARGIN:
                best, by = name, "error"
            elif abs(ea - eb) <= ERROR_MARGIN and a["latency_p50"] - b["latency_p50"] > LATENCY_MARGIN:
                best, by = name, "latency"
        winners[axis] = {"winner": best, "by": by, "baseline": base}
    return winners


def to_markdown(summary: dict, winners: dict) -> str:
    lines = ["| axis | setting | n | error mean (units) | misses | latency p50 / p95 (ms) | drops | result |", "|---|---|---|---|---|---|---|---|"]
    for axis, settings in summary.items():
        for name, s in settings.items():
            tag = f"winner ({winners[axis]['by']})" if winners[axis]["winner"] == name else ""
            lines.append(f"| {axis} | {name} | {s['n']} | {s['error_mean']} | {s['misses']} | {s['latency_p50']:.0f} / {s['latency_p95']:.0f} | {s['drops']} | {tag} |")
    return "\n".join(lines) + "\n"
```

```python
@mcp.tool()
async def bench(axes: list[str] = ["aim", "pace", "coex"], taps_per_setting: int = 5) -> str:
    """A/B bench: with a blank Freeform board open and its pen selected, taps a planned grid per setting,
    reads the dots back, and ranks settings by tap error then latency. Axes: aim, pace, coex (live) and res,
    fps (need a mirroring restart per setting). Writes ~/.esp32-agent/bench/<date>.md and trial telemetry.
    Restores the baseline at the end and presses Home."""
    png, lag = await link.frame()
    if not lag["live"] or find_dots(Frame.from_jpeg(png)) is None:
        return "refused: open a blank Freeform board (white canvas) with the pen tool selected, mirroring live"
    trials: list[Trial] = []
    try:
        for axis in axes:
            for value in BENCH_AXES[axis]:
                await _apply_setting(axis, value)
                before = Frame.from_jpeg((await link.frame())[0])
                drops0 = int(link.status.get("ble", {}).get("drop", 0))
                lat = []
                for (x, y) in BENCH_PLAN[:taps_per_setting]:
                    t0 = time.monotonic()
                    await phone.tap(x, y, precise=(value == "precise") if axis == "aim" else False)
                    await link._fresh_frame(link.board_now())
                    lat.append((time.monotonic() - t0) * 1000)
                after = Frame.from_jpeg((await link.frame())[0])
                found = find_dots(after) or []
                known = find_dots(before) or []
                new = [d for d in found if all(abs(d[0] - k[0]) + abs(d[1] - k[1]) > 20 for k in known)]
                errors = match_dots(BENCH_PLAN[:taps_per_setting], new)
                drops = int(link.status.get("ble", {}).get("drop", 0)) - drops0
                for e, l in zip(errors, lat):
                    tr = Trial(axis, value, e, round(l), drops)
                    trials.append(tr)
                    TELEMETRY.event(**tr.row())
                await link.cmd("key cmd+z")  # Freeform: undo the dots (one per tap)
                for _ in range(taps_per_setting - 1):
                    await link.cmd("key cmd+z")
            # baseline restore for this axis
            await _apply_setting(axis, BENCH_AXES[axis][0])
    finally:
        await phone.edge_home()
    summary = summarize(trials)
    winners = pick_winners(summary)
    out = HOME / "bench"
    out.mkdir(parents=True, exist_ok=True)
    path = out / (time.strftime("%Y-%m-%d-%H%M") + ".md")
    path.write_text(f"# Bench {time.strftime('%Y-%m-%d %H:%M')}\n\n" + to_markdown(summary, winners))
    return json.dumps({"summary": summary, "winners": winners, "file": str(path)}, default=str)
```
